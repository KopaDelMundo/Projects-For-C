#include <stdio.h>

int main(void)
{
    //read an entire file (fgets)
    FILE *fp;
    char s[1024]; //should be big enough for any line this program would get
    int linecount = 0;

    fp = fopen("hello.txt", "r");

    while(fgets(s, sizeof s, fp) != NULL)
        printf("%d: %s", ++linecount, s);
    
    fclose(fp);

}

#include <stdio.h>

int main(void)
{
    FILE *fp;
    char name[1024];
    float length;
    int mass;

    fp = fopen("whales.txt", "r");

    while(fscanf(fp, "%s %f %d", name, &length, &mass) != EOF)
    {
        printf("%s whale, %d tonnes, %.1f meters\n", name, mass, length);
    }

    fclose(fp);
}

binary read

#include <stdio.h>

int main(void)
{
    FILE *fp;
    unsigned char c;

    fp = fopen("output.bin", "rb"); //rb for "read binary"

    while (fread(&c, sizeof(char), 1, fp) > 0)
    {
        printf("%d\n", c);
    }

    fclose(fp);
}
//appears as ints because we are asking them to be read as such in printf()
